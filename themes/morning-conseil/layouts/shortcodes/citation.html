{{- /* Citation de l'expert redacteur. Usage sans parametre : l'auteur de l'article est repris automatiquement. {{< citation >}}Le texte de la citation.{{< /citation >}} Pour citer quelqu'un d'autre : {{< citation auteur="Prenom Nom" role="Fonction" >}}...{{< /citation >}} */ -}} {{- $lang := .Page.Site.Language.Lang -}} {{- $author := index .Page.Site.Data.authors (.Page.Params.author | default "") -}} {{- $name := .Get "auteur" -}} {{- $role := .Get "role" -}} {{- $avatar := "" -}} {{- $url := "" -}} {{- if and (not $name) $author -}} {{- $name = $author.name -}} {{- $role = index $author.role $lang -}} {{- $avatar = $author.avatar -}} {{- $url = printf "authors/%s/" $author.id | relLangURL -}} {{- end -}}
